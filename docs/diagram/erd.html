    users {

        string password_digest

    }

    sessions {
        bigint id PK
        bigint user_id FK
        string ip_address
        string user_agent
    }

    events {
        bigint id PK
        string name
        date start_date
        date end_date
        integer target_cardio "default 24"
        integer target_strength "default 16"
        integer total_weeks "default 4"
        integer daily_point_cap "default 4"
        integer streak_penalty_per_violation "default 25"
        decimal long_run_target_km "default 10"
        integer weekly_cardio_target "default 2"
        integer weekly_strength_target "default 1"
        integer bonus_cardio_target "default 3"
        integer bonus_strength_target "default 2"
        integer max_consecutive_cardio_days "default 3"
        integer fun_sport_point_target "default 4"
    }

    criteria {
        bigint id PK
        string code UK "C1 sampai C10"
        string name
        decimal weight "total harus 1,0"
        integer criterion_type "0 benefit, 1 cost"
        integer position UK
    }

    participants {
        bigint id PK
        bigint event_id FK
        bigint user_id FK "boleh kosong"
        string nip UK
        string name
        string department
        string alternative_code UK "A1 sampai An"
    }

    activity_logs {
        bigint id PK
        bigint participant_id FK
        date activity_date
        integer activity_type "0 cardio, 1 strength, 2 long_run, 3 fun_sport"
        decimal raw_points
        decimal distance_km
        string evidence_url
        boolean evidence_valid
        integer source "0 manual, 1 import"
        uuid import_batch_id "penanda batch UC-05"
    }

    criterion_scores {
        bigint id PK
        bigint participant_id FK
        bigint criterion_id FK
        decimal raw_value
        decimal normalized_value "skala 0 sampai 100"
        string notes "catatan evaluasi UC-09"
    }

    topsis_runs {
        bigint id PK
        bigint event_id FK
        bigint executed_by_id FK "users, boleh kosong"
        datetime executed_at
        jsonb weights_snapshot
        jsonb decision_matrix "matriks X"
        jsonb normalized_matrix "matriks R"
        jsonb weighted_matrix "matriks Y"
        jsonb ideal_positive "solusi A+"
        jsonb ideal_negative "solusi A-"
    }

    ranking_results {
        bigint id PK
        bigint topsis_run_id FK
        bigint participant_id FK
        decimal d_positive "jarak ke A+"
        decimal d_negative "jarak ke A-"
        decimal preference_value "nilai Vi"
        integer rank
    }
